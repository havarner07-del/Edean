BINOM - TI-BASIC version for the older black-and-white TI-84 Plus
Expands (A*U^P + B*V^Q)^N
Type 0 for U or V if that part is just a number.
("->" means the STO-> key)

ClrHome
Disp "(AU^P+BV^Q)^N","INTEGERS ONLY","VAR 0=NO VAR"
Input "A=",A
Input "U=",Str1
Input "P=",P
Input "B=",B
Input "V=",Str2
Input "Q=",Q
Input "N=",N
If fPart(A) or fPart(B) or fPart(N) or A=0 or B=0 or N<0
Then
Disp "BAD INPUT"
Stop
End
ClrHome
For(K,0,N)
(N nCr K)*A^(N-K)*B^K->C
P*(N-K)->E
Q*K->F
If C<0
Then
"-"->Str3
Else
"+"->Str3
End
(Str1≠"0" and E≠0) or (Str2≠"0" and F≠0)->G
{abs(C),E,F}->L3
For(J,1,3)
L3(J)->X
"0"->Str9
If X>0
Then
" "->Str9
While X>0
sub("0123456789",round(X-10*int(X/10),0)+1,1)+Str9->Str9
int(X/10)->X
End
sub(Str9,1,length(Str9)-1)->Str9
End
If J=1 and (abs(C)≠1 or not(G))
Str3+Str9->Str3
If J=2 and Str1≠"0" and E≠0
Then
Str3+Str1->Str3
If E>1
Str3+"^"+Str9->Str3
End
If J=3 and Str2≠"0" and F≠0
Then
Str3+Str2->Str3
If F>1
Str3+"^"+Str9->Str3
End
End
Disp Str3
If fPart((K+1)/6)=0 and K<N
Pause
End
